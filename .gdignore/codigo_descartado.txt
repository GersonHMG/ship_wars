		if desired.angle_to(direction) < 0 :		#Rotacion horaria
			rotation += turn_speed*delta
		if desired.angle_to(direction) > 0:			#Rotacion anti-horaria

func eliptic_path(delta):
	if node:
		var target = get_node(node)
		var point = (self.position - target.position)
		var d = 200
		var a = 100
		
		var m = Transform2D()
		m = m.translated(target.position)
		var distance = (target.position).distance_to(self.position)
		if distance > d-a and distance < d+a:
			$CanvasLayer/action.text = "eliptic"
			var angle = point.angle_to(Vector2(1,0))
			var next = Vector2(d*sin(angle),d*cos(angle))
			next = m.xform(next)
			follow_target(delta, next)
		else:
			point = point.clamped(d)
			point = m.xform(point)
			$CanvasLayer/action.text = "to_point"
			follow_target(delta, point)



func follow_target(delta, point):
	engine_power = max_engine_power
	var desired = (point - self.position).normalized()
	var distance = self.position.distance_to(point)
	var steer = (desired*engine_power - velocity).normalized()*5
	head = Vector2(1,0).rotated(rotation).normalized()
	var t = velocity.length()/abs(breaking_power)
	if( velocity.length() * t/2) >= distance or distance < 30 :
		engine_power = 0
		print("Detenido")
		breaks(delta)
	else:
		rotation -= desired.angle_to(head)*delta*steer.length()
		acceleration = head*engine_power


func free_route():
	n.randomize()
	var probabilidad = n.randi_range(1,2)
	if probabilidad == 2:
		var degree = n.randf_range(-PI/6,PI/6)
		return to_point(ship.position + Vector2(1,0).rotated(ship.rotation+degree)*5)
	return to_point(ship.position + Vector2(1,0).rotated(ship.rotation)*5)


func seek(target):
	return to_point(target.global_position)

func to_point(point):
	var desired = (point - ship.position).normalized()
	var steering = (desired*ship.speed-ship.velocity).normalized()
	return steering
